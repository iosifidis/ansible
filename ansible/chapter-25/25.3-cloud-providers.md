# 25.3 Cloud Providers ☁️

## Proxmox Plugin

```bash
# Εγκατάσταση
ansible-galaxy collection install community.general
```

```yaml
# inventory/proxmox.yml
---
plugin: community.general.proxmox

# Connection
url:      "https://pve.example.com:8006"
user:     "ansible@pam"
password: "{{ lookup('env', 'PROXMOX_PASSWORD') }}"
validate_certs: false

# Φίλτρα
want_facts: true
exclude_state: ["stopped", "paused"]    # ← μόνο running VMs

# Groups
group_prefix: "proxmox_"

# Variables
compose:
  ansible_host: proxmox_ipconfig0 | regex_replace('ip=([^,/]+).*', '\\1')
  ansible_user: "ansible"
```

## AWS EC2 Plugin

```bash
ansible-galaxy collection install amazon.aws
pip install boto3
```

```yaml
# inventory/aws-ec2.yml
---
plugin: amazon.aws.aws_ec2
regions:
  - eu-west-1
  - eu-central-1

filters:
  instance-state-name: running
  "tag:Environment": production
  "tag:ManagedBy":   ansible

keyed_groups:
  - key:    tags.Role
    prefix: role
  - key:    placement.region
    prefix: region

compose:
  ansible_host: public_ip_address
  ansible_user: "ansible"

hostnames:
  - tag:Name          # ← χρήση tag Name ως hostname

cache:
  enabled: true
  connection: /tmp/aws_cache
  timeout: 300
```

```bash
# AWS credentials
export AWS_ACCESS_KEY_ID=...
export AWS_SECRET_ACCESS_KEY=...
export AWS_DEFAULT_REGION=eu-west-1

ansible-inventory -i inventory/aws-ec2.yml --graph
```
