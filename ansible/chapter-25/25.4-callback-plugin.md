# 25.4 Custom Callback Plugin 📢

## Τι είναι Callback Plugin

Τα **callback plugins** αντικαθιστούν / επεκτείνουν το default output της Ansible. Χρήσιμα για:
- Custom formatting
- Αποστολή αποτελεσμάτων σε external systems
- Metrics collection

---

## Minimal callback plugin

```python
# callback_plugins/notify_slack.py
# Στέλνει Slack notification όταν play τελειώνει

from ansible.plugins.callback import CallbackBase
import urllib.request
import json
import os

DOCUMENTATION = '''
callback: notify_slack
type: notification
short_description: Send Slack notification on play end
options:
  webhook_url:
    description: Slack webhook URL
    env:
      - name: SLACK_WEBHOOK
    ini:
      - section: callback_notify_slack
        key: webhook_url
'''


class CallbackModule(CallbackBase):
    CALLBACK_VERSION    = 2.0
    CALLBACK_TYPE       = 'notification'
    CALLBACK_NAME       = 'notify_slack'
    CALLBACK_NEEDS_ENABLED = True

    def __init__(self):
        super().__init__()
        self.webhook_url = os.environ.get('SLACK_WEBHOOK', '')
        self.play_name   = ''
        self.stats       = {}

    def v2_playbook_on_play_start(self, play):
        self.play_name = play.get_name()

    def v2_playbook_on_stats(self, stats):
        """Καλείται στο τέλος — αποστολή σύνοψης"""
        if not self.webhook_url:
            return

        hosts = sorted(stats.processed.keys())
        summary = []
        ok_total = failed_total = 0

        for host in hosts:
            s = stats.summarize(host)
            ok_total     += s['ok']
            failed_total += s['failures']
            status = '✅' if s['failures'] == 0 else '❌'
            summary.append(
                f"{status} *{host}*: ok={s['ok']} "
                f"changed={s['changed']} failed={s['failures']}"
            )

        emoji   = '✅' if failed_total == 0 else '❌'
        message = {
            'text': (
                f"{emoji} Ansible: *{self.play_name}*\n" +
                '\n'.join(summary)
            )
        }

        try:
            data = json.dumps(message).encode()
            req  = urllib.request.Request(
                self.webhook_url,
                data=data,
                headers={'Content-Type': 'application/json'}
            )
            urllib.request.urlopen(req, timeout=5)
        except Exception as e:
            self._display.warning(f'Slack notification failed: {e}')
```

## Ενεργοποίηση

```ini
# ansible.cfg
[defaults]
callback_plugins  = ./callback_plugins
callbacks_enabled = notify_slack
```

```bash
# Εκτέλεση με Slack notification
SLACK_WEBHOOK=https://hooks.slack.com/... \
ansible-playbook site.yml --limit nextcloud
```

```
Callback Plugin
```

# 25.5 Πρακτικό παράδειγμα — Custom Module για API

```python
# library/netbox_service.py
# Διαχειρίζεται services στο NetBox CMDB

from ansible.module_utils.basic import AnsibleModule
from ansible.module_utils.urls import fetch_url
import json


def get_service(module, url, headers, name):
    resp, info = fetch_url(
        module, f'{url}/api/ipam/services/?name={name}',
        headers=headers, method='GET'
    )
    if info['status'] == 200:
        data = json.loads(resp.read())
        return data['results'][0] if data['count'] > 0 else None
    return None


def main():
    module = AnsibleModule(
        argument_spec=dict(
            netbox_url   = dict(type='str', required=True),
            netbox_token = dict(type='str', required=True, no_log=True),
            name         = dict(type='str', required=True),
            port         = dict(type='int', required=True),
            protocol     = dict(type='str', default='tcp',
                                choices=['tcp', 'udp']),
            virtual_machine = dict(type='str', required=True),
            state        = dict(type='str', default='present',
                                choices=['present', 'absent']),
        ),
        supports_check_mode=True,
    )

    p      = module.params
    url    = p['netbox_url'].rstrip('/')
    headers = {
        'Authorization': f"Token {p['netbox_token']}",
        'Content-Type':  'application/json',
        'Accept':        'application/json',
    }

    existing = get_service(module, url, headers, p['name'])

    if p['state'] == 'present' and existing:
        module.exit_json(changed=False, service=existing)

    if p['state'] == 'absent' and not existing:
        module.exit_json(changed=False)

    if module.check_mode:
        module.exit_json(changed=True)

    if p['state'] == 'present':
        data = json.dumps({
            'name': p['name'],
            'port': p['port'],
            'protocol': p['protocol'],
        }).encode()
        resp, info = fetch_url(
            module, f'{url}/api/ipam/services/',
            headers=headers, method='POST', data=data
        )
        if info['status'] == 201:
            module.exit_json(changed=True, service=json.loads(resp.read()))
        else:
            module.fail_json(msg=f"API error: {info['msg']}")

    else:
        fetch_url(
            module,
            f"{url}/api/ipam/services/{existing['id']}/",
            headers=headers, method='DELETE'
        )
        module.exit_json(changed=True)


if __name__ == '__main__':
    main()
```

```yaml
# Χρήση στο playbook
tasks:
  - name: Register service in NetBox
    netbox_service:
      netbox_url:      "https://netbox.example.com"
      netbox_token:    "{{ vault_netbox_token }}"
      name:            "myapp-api"
      port:            8080
      protocol:        tcp
      virtual_machine: nextcloud
      state:           present
    register: nb_service
    no_log: true
```
