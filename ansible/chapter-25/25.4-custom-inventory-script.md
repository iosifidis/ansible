# 25.4 Custom Inventory Script 🐍

## Πότε custom script

Όταν δεν υπάρχει διαθέσιμο plugin (π.χ. proprietary CMDB):

```python
#!/usr/bin/env python3
# inventory/my_cmdb.py

"""
Custom inventory script για εσωτερικό CMDB.
Χρήση: ansible-playbook site.yml -i inventory/my_cmdb.py
"""

import json
import sys
import os
import urllib.request


def get_hosts_from_cmdb():
    """Ανάκτηση hosts από εσωτερικό CMDB API"""
    api_url   = os.environ.get('CMDB_URL', 'https://cmdb.internal/api')
    api_token = os.environ.get('CMDB_TOKEN', '')

    req = urllib.request.Request(
        f'{api_url}/hosts?status=active',
        headers={'Authorization': f'Bearer {api_token}'}
    )
    response = urllib.request.urlopen(req, timeout=10)
    return json.loads(response.read())


def build_inventory(hosts):
    """Δημιουργία Ansible inventory structure"""
    inventory = {
        '_meta': {
            'hostvars': {}    # ← μεταβλητές ανά host
        }
    }

    for host in hosts:
        hostname = host['name']
        role     = host.get('role', 'ungrouped')
        env      = host.get('environment', 'production')

        # ── Προσθήκη σε groups ─────────────────
        for group in [role, env, f"{role}_{env}"]:
            if group not in inventory:
                inventory[group] = {'hosts': []}
            inventory[group]['hosts'].append(hostname)

        # ── Host variables ────────────────────
        inventory['_meta']['hostvars'][hostname] = {
            'ansible_host': host.get('ip_address', hostname),
            'ansible_user': host.get('ansible_user', 'ansible'),
            'ansible_port': host.get('ssh_port', 22),
            'cmdb_id':      host.get('id'),
            'cmdb_tags':    host.get('tags', []),
        }

    return inventory


def main():
    # Ansible καλεί με --list ή --host <hostname>
    if len(sys.argv) == 2 and sys.argv[1] == '--list':
        hosts     = get_hosts_from_cmdb()
        inventory = build_inventory(hosts)
        print(json.dumps(inventory, indent=2))

    elif len(sys.argv) == 3 and sys.argv[1] == '--host':
        # Επιστρέφει hostvars για συγκεκριμένο host
        # (χρησιμοποιούμε _meta παραπάνω, οπότε αδειο)
        print(json.dumps({}))

    else:
        print(f"Usage: {sys.argv[0]} --list | --host <hostname>",
              file=sys.stderr)
        sys.exit(1)


if __name__ == '__main__':
    main()
```

```bash
# Δικαιώματα εκτέλεσης
chmod +x inventory/my_cmdb.py

# Test
CMDB_URL=https://cmdb.internal CMDB_TOKEN=abc123 \
./inventory/my_cmdb.py --list | jq .

# Χρήση
CMDB_TOKEN=abc123 \
ansible-playbook site.yml -i inventory/my_cmdb.py
```
