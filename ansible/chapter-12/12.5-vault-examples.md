# 12.5 Πρακτικά παραδείγματα Ansible Vault 🛠️

## Τι θα φτιάξουμε

```
Ένα πλήρες, real-world setup με Vault:

├── Δομή αρχείων με secrets
├── Playbook που χρησιμοποιεί secrets
├── Templates με sensitive data
└── Διαχείριση secrets ανά environment
```

---

## Βήμα 1: Δομή αρχείων

```bash
# Δημιουργία δομής
mkdir -p ~/ansible/inventory/group_vars/all
mkdir -p ~/ansible/vars
mkdir -p ~/ansible/vars/environments

# Η τελική δομή:
tree ~/ansible/inventory/group_vars/
# group_vars/
# └── all/
#     ├── vars.yml    ← plaintext
#     └── vault.yml   ← κρυπτογραφημένο
```

## Βήμα 2: Plaintext Variables

```bash
cat > ~/ansible/inventory/group_vars/all/vars.yml << 'EOF'
---
# ============================================================
# Global Variables — PLAINTEXT
# ============================================================

# ── System ────────────────────────────────────
system_timezone: "Europe/Athens"
system_locale:   "el_GR.UTF-8"

# ── SSH ───────────────────────────────────────
ssh_port:           2022
ssh_allowed_users:
  - ansible

# ── App ───────────────────────────────────────
app_name:    myapp
app_env:     production
http_port:   80
https_port:  443

# ── Database ──────────────────────────────────
db_host:     localhost
db_port:     5432
db_name:     "{{ app_name }}_db"
db_user:     "{{ app_name }}_user"

# ── References to vault vars ──────────────────
# Convention: plaintext var → vault var
db_password:        "{{ vault_db_password }}"
db_root_password:   "{{ vault_db_root_password }}"
smtp_password:      "{{ vault_smtp_password }}"
api_github_token:   "{{ vault_api_github_token }}"
ssl_key_password:   "{{ vault_ssl_key_password }}"
EOF
```

## Βήμα 3: Vault Variables

```bash
# Δημιουργία plaintext πρώτα
cat > /tmp/vault_plain.yml << 'EOF'
---
# ============================================================
# Vault Variables — ΘΑ ΚΡΥΠΤΟΓΡΑΦΗΘΟΥΝ
# ============================================================

# ── Database ──────────────────────────────────
vault_db_password:      "DBPass@2026#Secure!"
vault_db_root_password: "RootPass@2026#Super!"

# ── SMTP ──────────────────────────────────────
vault_smtp_password:    "EmailPass@2026!"

# ── API Tokens ────────────────────────────────
vault_api_github_token: "ghp_abc123def456xyz789"

# ── SSL ───────────────────────────────────────
vault_ssl_key_password: "SSLKey@2026#Pass!"
EOF

# Αντιγραφή και κρυπτογράφηση
cp /tmp/vault_plain.yml \
   ~/ansible/inventory/group_vars/all/vault.yml

ansible-vault encrypt \
    ~/ansible/inventory/group_vars/all/vault.yml

# Επαλήθευση
ansible-vault view \
    ~/ansible/inventory/group_vars/all/vault.yml

# Καθαρισμός plaintext
rm /tmp/vault_plain.yml
```

## Βήμα 4: Secrets Playbook

```bash
vim ~/ansible/playbooks/secrets-demo.yml
```

```yaml
---
# ============================================================
# Secrets Demo Playbook
# Χρήση Ansible Vault σε real-world scenario
# ============================================================

- name: Application Secrets Deployment
  hosts: "{{ target | default('all_managed') }}"
  become: true
  gather_facts: true

  vars:
    # Extra vars (δεν είναι secrets)
    app_user:    "{{ app_name }}"
    app_dir:     "/opt/{{ app_name }}"
    config_dir:  "/etc/{{ app_name }}"
    log_dir:     "/var/log/{{ app_name }}"

  tasks:

    # ══════════════════════════════════════════
    # ΠΡΟΕΤΟΙΜΑΣΙΑ
    # ══════════════════════════════════════════

    - name: Εμφάνιση deployment info
      ansible.builtin.debug:
        msg:
          - "━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━"
          - "App    : {{ app_name }}"
          - "Env    : {{ app_env }}"
          - "Host   : {{ inventory_hostname }}"
          - "DB     : {{ db_host }}:{{ db_port }}/{{ db_name }}"
          - "Secrets: *** PROTECTED BY VAULT ***"
          - "━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━"

    - name: Δημιουργία app directories
      ansible.builtin.file:
        path:  "{{ item }}"
        state: directory
        mode:  '0750'
        owner: root
        group: root
      loop:
        - "{{ app_dir }}"
        - "{{ config_dir }}"
        - "{{ log_dir }}"

    - name: Δημιουργία app user
      ansible.builtin.user:
        name:        "{{ app_user }}"
        shell:       /bin/false
        create_home: false
        system:      true
        state:       present

    # ══════════════════════════════════════════
    # DATABASE CONFIGURATION
    # ══════════════════════════════════════════

    - name: Εγκατάσταση PostgreSQL client
      ansible.builtin.package:
        name:  "{{ 'postgresql-client' if ansible_facts['os_family'] == 'Debian'
                   else 'postgresql' }}"
        state: present

    - name: Deploy database config
      ansible.builtin.template:
        src:   templates/db.conf.j2
        dest:  "{{ config_dir }}/db.conf"
        owner: "{{ app_user }}"
        group: "{{ app_user }}"
        mode:  '0600'          # ← μόνο ο app user!
      no_log: true             # ← αποκρύπτει password

    - name: Έλεγχος database connectivity
      ansible.builtin.command:
        cmd: >
          pg_isready
          -h {{ db_host }}
          -p {{ db_port }}
          -U {{ db_user }}
      register:     db_check
      changed_when: false
      ignore_errors: true
      no_log: true

    - name: Database status
      ansible.builtin.debug:
        msg: >
          Database:
          {{ '✅ Accessible' if db_check.rc == 0
             else '❌ Not accessible (OK if not installed)' }}

    # ══════════════════════════════════════════
    # APPLICATION CONFIGURATION
    # ══════════════════════════════════════════

    - name: Deploy app environment file
      ansible.builtin.template:
        src:   templates/app_env.j2
        dest:  "{{ app_dir }}/.env"
        owner: "{{ app_user }}"
        group: "{{ app_user }}"
        mode:  '0600'
      no_log: true

    - name: Deploy SMTP config
      ansible.builtin.template:
        src:   templates/smtp.conf.j2
        dest:  "{{ config_dir }}/smtp.conf"
        owner: "{{ app_user }}"
        group: "{{ app_user }}"
        mode:  '0600'
      no_log: true

    # ══════════════════════════════════════════
    # API CONFIGURATION
    # ══════════════════════════════════════════

    - name: Deploy API config
      ansible.builtin.template:
        src:   templates/api.conf.j2
        dest:  "{{ config_dir }}/api.conf"
        owner: "{{ app_user }}"
        group: "{{ app_user }}"
        mode:  '0600'
      no_log: true

    # ══════════════════════════════════════════
    # PERMISSIONS & SECURITY
    # ══════════════════════════════════════════

    - name: Σωστά permissions σε config files
      ansible.builtin.file:
        path:    "{{ config_dir }}"
        owner:   "{{ app_user }}"
        group:   "{{ app_user }}"
        mode:    '0750'
        recurse: true

    - name: Έλεγχος sensitive files permissions
      ansible.builtin.stat:
        path: "{{ item }}"
      register: file_perms
      loop:
        - "{{ config_dir }}/db.conf"
        - "{{ app_dir }}/.env"
        - "{{ config_dir }}/smtp.conf"
        - "{{ config_dir }}/api.conf"
      loop_control:
        label: "{{ item }}"

    - name: Permissions report
      ansible.builtin.debug:
        msg: >
          {{ item.item }}:
          mode={{ item.stat.mode | default('N/A') }}
          {{ '✅' if item.stat.mode | default('') == '0100600'
             else '⚠️ Check permissions!' }}
      loop: "{{ file_perms.results }}"
      loop_control:
        label: "{{ item.item }}"

    # ══════════════════════════════════════════
    # ΤΕΛΙΚΟ REPORT
    # ══════════════════════════════════════════

    - name: Deployment report
      ansible.builtin.debug:
        msg:
          - "━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━"
          - "✅ DEPLOYMENT COMPLETE"
          - "━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━"
          - "App dir    : {{ app_dir }}"
          - "Config dir : {{ config_dir }}"
          - "Log dir    : {{ log_dir }}"
          - "Secrets    : Protected by Vault 🔐"
          - "━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━"
```

## Βήμα 5: Templates με Secrets

```bash
# Δημιουργία templates
mkdir -p ~/ansible/templates
```

```jinja
{# templates/db.conf.j2 #}
{# Database Configuration — Contains sensitive data! #}
[database]
host     = {{ db_host }}
port     = {{ db_port }}
name     = {{ db_name }}
user     = {{ db_user }}
password = {{ db_password }}

[pool]
min_conn = {{ db_pool_min | default(2)  }}
max_conn = {{ db_pool_max | default(10) }}
timeout  = {{ db_timeout  | default(30) }}
```

```jinja
{# templates/app_env.j2 #}
# Application Environment
# Generated by Ansible — DO NOT EDIT!

APP_NAME={{ app_name }}
APP_ENV={{ app_env }}
APP_DEBUG={{ 'false' if app_env == 'production' else 'true' }}
APP_KEY={{ vault_api_github_token | hash('sha256') }}

DB_HOST={{ db_host }}
DB_PORT={{ db_port }}
DB_DATABASE={{ db_name }}
DB_USERNAME={{ db_user }}
DB_PASSWORD={{ db_password }}

MAIL_HOST={{ smtp_host | default('smtp.gmail.com') }}
MAIL_PORT={{ smtp_port | default(587) }}
MAIL_PASSWORD={{ smtp_password }}

GITHUB_TOKEN={{ api_github_token }}
```

```jinja
{# templates/smtp.conf.j2 #}
# SMTP Configuration
[smtp]
host     = {{ smtp_host     | default('smtp.gmail.com') }}
port     = {{ smtp_port     | default(587)              }}
user     = {{ smtp_user     | default(app_name)         }}
password = {{ smtp_password }}
tls      = {{ 'true' if smtp_port | default(587) | int == 587 else 'false' }}
```

```jinja
{# templates/api.conf.j2 #}
# API Configuration
[github]
token    = {{ api_github_token }}
org      = {{ github_org | default('myorg') }}

{% if vault_ssl_key_password is defined %}
[ssl]
key_password = {{ ssl_key_password }}
{% endif %}
```

## Βήμα 6: Εκτέλεση

```bash
# ── Syntax check ──────────────────────────────
ansible-playbook playbooks/secrets-demo.yml \
    --syntax-check

# ── List tasks ────────────────────────────────
ansible-playbook playbooks/secrets-demo.yml \
    --list-tasks

# ── Dry run ───────────────────────────────────
ansible-playbook playbooks/secrets-demo.yml \
    --check \
    --limit nextcloud

# ── Εκτέλεση ──────────────────────────────────
ansible-playbook playbooks/secrets-demo.yml \
    --limit nextcloud \
    -v

# ── Επαλήθευση αρχείων στον server ───────────
ansible nextcloud -m ansible.builtin.command \
    -a "ls -la /etc/myapp/" \
    --become

ansible nextcloud -m ansible.builtin.command \
    -a "stat /etc/myapp/db.conf" \
    --become
```

## Βήμα 7: Διαχείριση Secrets ανά Environment

```bash
# ── Production secrets ────────────────────────
cat > /tmp/prod_vault.yml << 'EOF'
---
vault_db_password:      "ProdDBPass@2026#Ultra!"
vault_db_root_password: "ProdRootPass@2026#Super!"
vault_smtp_password:    "ProdEmail@2026!"
vault_api_github_token: "ghp_PROD_abc123"
vault_ssl_key_password: "ProdSSL@2026!"
EOF

cp /tmp/prod_vault.yml ~/ansible/vars/environments/production.yml
ansible-vault encrypt ~/ansible/vars/environments/production.yml
rm /tmp/prod_vault.yml

# ── Staging secrets ───────────────────────────
cat > /tmp/staging_vault.yml << 'EOF'
---
vault_db_password:      "StagingDBPass@2026!"
vault_db_root_password: "StagingRoot@2026!"
vault_smtp_password:    "StagingEmail@2026!"
vault_api_github_token: "ghp_STAGING_xyz789"
vault_ssl_key_password: "StagingSSL@2026!"
EOF

cp /tmp/staging_vault.yml ~/ansible/vars/environments/staging.yml
ansible-vault encrypt ~/ansible/vars/environments/staging.yml
rm /tmp/staging_vault.yml
```

```yaml
# playbooks/secrets-demo.yml — με environment support
- name: Application Secrets Deployment
  hosts: all
  become: true

  vars_files:
    - "vars/environments/{{ app_env | default('production') }}.yml"
    # ↑ φορτώνει production.yml ή staging.yml αυτόματα!
```

```bash
# Εκτέλεση για production
ansible-playbook playbooks/secrets-demo.yml \
    -e "app_env=production" \
    --limit nextcloud

# Εκτέλεση για staging
ansible-playbook playbooks/secrets-demo.yml \
    -e "app_env=staging" \
    --limit nextcloud
```

## Vault Maintenance Script

```bash
# ~/ansible/scripts/vault-maintenance.sh
cat > ~/ansible/scripts/vault-maintenance.sh << 'EOF'
#!/bin/bash
# Vault Maintenance Script

ANSIBLE_DIR=~/ansible
VAULT_FILES=(
    "inventory/group_vars/all/vault.yml"
    "vars/environments/production.yml"
    "vars/environments/staging.yml"
)

echo "━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━"
echo "Ansible Vault Maintenance"
echo "━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━"

case "$1" in

    view)
        echo "Viewing: $2"
        ansible-vault view "${ANSIBLE_DIR}/$2"
        ;;

    edit)
        echo "Editing: $2"
        ansible-vault edit "${ANSIBLE_DIR}/$2"
        ;;

    rekey)
        echo "Rekeying all vault files..."
        for f in "${VAULT_FILES[@]}"; do
            echo "  → ${f}"
            ansible-vault rekey "${ANSIBLE_DIR}/${f}"
        done
        echo "✅ Rekey complete!"
        ;;

    check)
        echo "Checking vault files..."
        for f in "${VAULT_FILES[@]}"; do
            if grep -q "ANSIBLE_VAULT" "${ANSIBLE_DIR}/${f}" 2>/dev/null; then
                echo "  ✅ ${f} → encrypted"
            else
                echo "  ❌ ${f} → NOT encrypted!"
            fi
        done
        ;;

    *)
        echo "Usage: $0 {view|edit|rekey|check} [file]"
        exit 1
        ;;
esac
EOF

chmod +x ~/ansible/scripts/vault-maintenance.sh

# Χρήση:
# ./scripts/vault-maintenance.sh check
# ./scripts/vault-maintenance.sh view inventory/group_vars/all/vault.yml
# ./scripts/vault-maintenance.sh rekey
```
