# 21.5 Firewall Management 🔥

## ufw — Debian/Ubuntu

```yaml
tasks:

  # ── Enable UFW ───────────────────────────
  - name: Enable UFW
    community.general.ufw:
      state: enabled
      policy: deny    # ← default deny all
    when: ansible_facts['os_family'] == "Debian"

  # ── Allow SSH (πάντα πρώτα!) ─────────────
  - name: Allow SSH
    community.general.ufw:
      rule:    allow
      port:    "{{ ansible_port | default(22) }}"
      proto:   tcp
      comment: "SSH access"
    when: ansible_facts['os_family'] == "Debian"

  # ── Allow services ────────────────────────
  - name: Allow web traffic
    community.general.ufw:
      rule:    allow
      port:    "{{ item.port }}"
      proto:   tcp
      comment: "{{ item.comment }}"
    loop:
      - { port: "80",   comment: "HTTP"  }
      - { port: "443",  comment: "HTTPS" }
      - { port: "8080", comment: "App"   }
    when: ansible_facts['os_family'] == "Debian"

  # ── Allow from specific IP ────────────────
  - name: Allow monitoring server
    community.general.ufw:
      rule:    allow
      src:     "192.168.1.50"    # ← μόνο από αυτό το IP
      port:    "9100"
      proto:   tcp
      comment: "Prometheus node exporter"
    when: ansible_facts['os_family'] == "Debian"

  # ── Deny specific port ────────────────────
  - name: Block Telnet
    community.general.ufw:
      rule:  deny
      port:  "23"
      proto: tcp
    when: ansible_facts['os_family'] == "Debian"
```

## firewalld — RedHat/openSUSE

```yaml
tasks:

  # ── Enable & Start firewalld ──────────────
  - name: Start firewalld
    ansible.builtin.service:
      name:    firewalld
      state:   started
      enabled: true
    when: ansible_facts['os_family'] in ["RedHat", "Suse"]

  # ── Allow service ─────────────────────────
  - name: Allow HTTP/HTTPS
    ansible.posix.firewalld:
      service:   "{{ item }}"
      permanent: true
      state:     enabled
    loop:
      - http
      - https
      - ssh
    when: ansible_facts['os_family'] in ["RedHat", "Suse"]
    notify: Reload firewalld

  # ── Allow port ────────────────────────────
  - name: Allow custom port
    ansible.posix.firewalld:
      port:      "8080/tcp"
      permanent: true
      state:     enabled
    when: ansible_facts['os_family'] in ["RedHat", "Suse"]
    notify: Reload firewalld

  # ── Rich rule (από συγκεκριμένο IP) ───────
  - name: Allow monitoring
    ansible.posix.firewalld:
      rich_rule: >
        rule family="ipv4"
        source address="192.168.1.50"
        port port="9100" protocol="tcp"
        accept
      permanent: true
      state:     enabled
    when: ansible_facts['os_family'] in ["RedHat", "Suse"]
    notify: Reload firewalld

handlers:
  - name: Reload firewalld
    ansible.builtin.service:
      name:  firewalld
      state: reloaded
```

## Cross-distro firewall management

```yaml
tasks:

  # ── UFW (Debian/Ubuntu) ──────────────────
  - name: Allow port (UFW)
    community.general.ufw:
      rule:  allow
      port:  "{{ item }}"
      proto: tcp
    loop: "{{ open_ports }}"
    when: ansible_facts['os_family'] == "Debian"

  # ── Firewalld (RedHat/openSUSE) ───────────
  - name: Allow port (firewalld)
    ansible.posix.firewalld:
      port:      "{{ item }}/tcp"
      permanent: true
      state:     enabled
    loop: "{{ open_ports }}"
    when: ansible_facts['os_family'] in ["RedHat", "Suse"]
    notify: Reload firewalld

  # ── Define ports στο vars ────────────────
  # vars:
  #   open_ports: [22, 80, 443, 8080]
```
