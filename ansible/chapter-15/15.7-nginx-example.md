# 15.7 Πρακτικό παράδειγμα — Nginx Role από το μηδέν 🛠️

## Τι θα φτιάξουμε

```
Ένα production-ready nginx role που:
├── Εγκαθιστά nginx (και στις 3 distros)
├── Ρυθμίζει βάσει system resources (facts)
├── Υποστηρίζει virtual hosts
├── Έχει validation
├── Έχει πλήρη tag support
└── Χρησιμοποιείται από playbook
```

---

## Βήμα 1: Δημιουργία & επαλήθευση δομής

```bash
# Μετάβαση στο roles directory
cd ~/ansible/roles

# Δημιουργία role (αν δεν υπάρχει ήδη)
ansible-galaxy role init nginx

# Επαλήθευση δομής
tree nginx/
```

## Βήμα 2: defaults/main.yml

```bash
cat > ~/ansible/roles/nginx/defaults/main.yml << 'EOF'
---
# ============================================================
# Nginx Role Defaults
# ============================================================

# ── Performance (auto-detected) ───────────────
nginx_worker_processes:   "{{ ansible_facts['processor_vcpus'] | default(1) | int }}"
nginx_worker_connections: >-
  {{ 1024 if ansible_facts['memtotal_mb'] | default(0) | int >= 2048 else 768 }}
nginx_keepalive_timeout:  65
nginx_client_max_body:    "10M"
nginx_server_tokens:      "off"

# ── Features ──────────────────────────────────
nginx_ssl_enabled:    false
nginx_gzip_enabled:   true
nginx_http2_enabled:  false

# ── Paths ─────────────────────────────────────
nginx_web_root:  /var/www/html
nginx_log_dir:   /var/log/nginx
nginx_conf_dir:  /etc/nginx

# ── Default vhost ─────────────────────────────
nginx_default_vhost: true
nginx_default_port:  80

# ── Virtual hosts ─────────────────────────────
nginx_vhosts: []
EOF
```

## Βήμα 5: tasks/validate.yml

```bash
cat > ~/ansible/roles/nginx/tasks/validate.yml << 'EOF'
---
# ============================================================
# Nginx Role — Validation
# ============================================================

- name: Validate nginx configuration variables
  ansible.builtin.assert:
    that:
      - nginx_default_port | int > 0
      - nginx_default_port | int < 65535
      - nginx_worker_connections | int > 0
      - nginx_web_root | regex_search('^/')
      - nginx_log_dir  | regex_search('^/')
      - not nginx_ssl_enabled or (nginx_ssl_cert is defined and nginx_ssl_cert | length > 0)
      - not nginx_ssl_enabled or (nginx_ssl_key  is defined and nginx_ssl_key  | length > 0)
    fail_msg: |
      ❌ Nginx role validation failed!
      Έλεγξε τις παρακάτω ρυθμίσεις:
      - nginx_default_port: {{ nginx_default_port }}
      - nginx_worker_connections: {{ nginx_worker_connections }}
      - nginx_web_root: {{ nginx_web_root }}
      - nginx_ssl_enabled: {{ nginx_ssl_enabled }}
    success_msg: "✅ Nginx role variables validated"

- name: Check OS support
  ansible.builtin.assert:
    that:
      - ansible_facts['os_family'] in ['Debian', 'RedHat', 'Suse']
    fail_msg: >
      ❌ OS {{ ansible_facts['os_family'] }} not supported!
      Supported: Debian, RedHat, Suse
EOF
```

## Βήμα 6: tasks/install.yml

```bash
cat > ~/ansible/roles/nginx/tasks/install.yml << 'EOF'
---
# ============================================================
# Nginx Role — Installation
# ============================================================

# ── Debian/Ubuntu ─────────────────────────────
- name: Update apt cache (Debian)
  ansible.builtin.apt:
    update_cache:     true
    cache_valid_time: 3600
  when: ansible_facts['os_family'] == "Debian"

- name: Install nginx (Debian)
  ansible.builtin.apt:
    name:  "{{ _nginx_packages['Debian'] }}"
    state: present
  when: ansible_facts['os_family'] == "Debian"

# ── RedHat/Fedora ─────────────────────────────
- name: Install nginx (RedHat)
  ansible.builtin.dnf:
    name:  "{{ _nginx_packages['RedHat'] }}"
    state: present
  when: ansible_facts['os_family'] == "RedHat"

# ── openSUSE ──────────────────────────────────
- name: Install nginx (openSUSE)
  community.general.zypper:
    name:  "{{ _nginx_packages['Suse'] }}"
    state: present
  when: ansible_facts['os_family'] == "Suse"

# ── Επαλήθευση ────────────────────────────────
- name: Verify nginx installation
  ansible.builtin.command:
    cmd: nginx -v
  register:     _nginx_ver
  changed_when: false

- name: Show nginx version
  ansible.builtin.debug:
    msg: "✅ {{ _nginx_ver.stderr | trim }}"
EOF
```

## Βήμα 7: tasks/configure.yml

```bash
cat > ~/ansible/roles/nginx/tasks/configure.yml << 'EOF'
---
# ============================================================
# Nginx Role — Configuration
# ============================================================

# ── Directories ───────────────────────────────
- name: Ensure directories exist
  ansible.builtin.file:
    path:  "{{ item.path }}"
    state: directory
    owner: "{{ item.owner | default('root') }}"
    group: "{{ item.group | default('root') }}"
    mode:  "{{ item.mode  | default('0755') }}"
  loop:
    - { path: "{{ nginx_log_dir }}", owner: www-data, group: adm }
    - { path: "{{ nginx_web_root }}", owner: www-data, group: www-data }
    - { path: "{{ nginx_conf_dir }}/conf.d" }
    - { path: "{{ _nginx_sites_available[ansible_facts['os_family']] }}" }
    - { path: "{{ _nginx_sites_enabled[ansible_facts['os_family']] }}" }
  loop_control:
    label: "{{ item.path }}"

# ── Main config ───────────────────────────────
- name: Deploy nginx.conf
  ansible.builtin.template:
    src:      nginx.conf.j2
    dest:     "{{ nginx_conf_dir }}/nginx.conf"
    owner:    root
    group:    root
    mode:     '0644'
    backup:   true
    validate: "{{ _nginx_validate }}"
  notify: Restart nginx

# ── Default page ──────────────────────────────
- name: Deploy default index.html
  ansible.builtin.template:
    src:   index.html.j2
    dest:  "{{ nginx_web_root }}/index.html"
    owner: www-data
    group: www-data
    mode:  '0644'
  when: nginx_default_vhost
EOF
```

## Βήμα 8: tasks/vhosts.yml

```bash
cat > ~/ansible/roles/nginx/tasks/vhosts.yml << 'EOF'
---
# ============================================================
# Nginx Role — Virtual Hosts
# ============================================================

- name: Deploy vhost configurations
  ansible.builtin.template:
    src:   vhost.conf.j2
    dest:  "{{ _nginx_sites_available[ansible_facts['os_family']] }}/{{ item.name }}.conf"
    owner: root
    group: root
    mode:  '0644'
    backup: true
  loop: "{{ nginx_vhosts }}"
  loop_control:
    label: "{{ item.name }}"
  notify: Reload nginx
  when: nginx_vhosts | length > 0

- name: Enable vhosts (Debian)
  ansible.builtin.file:
    src:   "{{ _nginx_sites_available['Debian'] }}/{{ item.name }}.conf"
    dest:  "{{ _nginx_sites_enabled['Debian'] }}/{{ item.name }}.conf"
    state: link
  loop: "{{ nginx_vhosts }}"
  loop_control:
    label: "{{ item.name }}"
  notify: Reload nginx
  when:
    - ansible_facts['os_family'] == "Debian"
    - nginx_vhosts | length > 0

- name: Remove default nginx site (Debian)
  ansible.builtin.file:
    path:  "{{ _nginx_sites_enabled['Debian'] }}/default"
    state: absent
  when:
    - ansible_facts['os_family'] == "Debian"
    - nginx_vhosts | length > 0
  notify: Reload nginx
EOF
```

## Βήμα 9: tasks/service.yml

```bash
cat > ~/ansible/roles/nginx/tasks/service.yml << 'EOF'
---
# ============================================================
# Nginx Role — Service Management
# ============================================================

- name: Enable and start nginx
  ansible.builtin.service:
    name:    "{{ _nginx_service }}"
    state:   started
    enabled: true

- name: Wait for nginx to be ready
  ansible.builtin.wait_for:
    port:    "{{ nginx_default_port }}"
    host:    localhost
    timeout: 30
    delay:   2

- name: Verify nginx responds
  ansible.builtin.uri:
    url:         "http://localhost:{{ nginx_default_port }}"
    status_code: [200, 301, 302]
  register:      _nginx_check
  ignore_errors: true

- name: Nginx service status
  ansible.builtin.debug:
    msg: >
      Nginx on port {{ nginx_default_port }}:
      {{ '✅ Running' if not _nginx_check.failed
         else '⚠️ Not responding — check manually' }}
EOF
```

## Βήμα 11: templates/nginx.conf.j2

```bash
cat > ~/ansible/roles/nginx/templates/nginx.conf.j2 << 'EOF'
{# Nginx Main Configuration #}
{# Managed by Ansible — DO NOT EDIT MANUALLY! #}
{# Role: nginx | Host: {{ inventory_hostname }} #}

user www-data;
worker_processes {{ nginx_worker_processes }};
pid /run/nginx.pid;
include /etc/nginx/modules-enabled/*.conf;

events {
    worker_connections {{ nginx_worker_connections }};
    multi_accept on;
}

http {
    # ── Basic ────────────────────────────────
    sendfile           on;
    tcp_nopush         on;
    tcp_nodelay        on;
    keepalive_timeout  {{ nginx_keepalive_timeout }};
    server_tokens      {{ nginx_server_tokens }};
    client_max_body_size {{ nginx_client_max_body }};

    # ── MIME types ────────────────────────────
    include      {{ nginx_conf_dir }}/mime.types;
    default_type application/octet-stream;

    # ── Logging ───────────────────────────────
    access_log {{ nginx_log_dir }}/access.log;
    error_log  {{ nginx_log_dir }}/error.log;

    # ── Gzip ─────────────────────────────────
    {% if nginx_gzip_enabled %}
    gzip            on;
    gzip_vary       on;
    gzip_proxied    any;
    gzip_comp_level 6;
    gzip_types
        text/plain
        text/css
        application/json
        application/javascript
        text/xml
        application/xml
        text/javascript;
    {% endif %}

    # ── Virtual Hosts ─────────────────────────
    include {{ nginx_conf_dir }}/conf.d/*.conf;
    {% if ansible_facts['os_family'] == 'Debian' %}
    include {{ nginx_conf_dir }}/sites-enabled/*;
    {% endif %}
}
EOF
```

## Βήμα 12: templates/vhost.conf.j2

```bash
cat > ~/ansible/roles/nginx/templates/vhost.conf.j2 << 'EOF'
{# Virtual Host: {{ item.name }} #}
{# Managed by Ansible — DO NOT EDIT MANUALLY! #}

server {
    listen      {{ item.port | default(80) }};
    {% if nginx_http2_enabled %}
    listen      {{ item.port | default(80) }} http2;
    {% endif %}
    server_name {{ item.domains | join(' ') }};
    root        {{ item.root | default(nginx_web_root) }};
    index       index.html index.htm index.php;

    # ── Logging ───────────────────────────────
    access_log {{ nginx_log_dir }}/{{ item.name }}_access.log;
    error_log  {{ nginx_log_dir }}/{{ item.name }}_error.log;

    # ── Security headers ──────────────────────
    add_header X-Frame-Options       "SAMEORIGIN"    always;
    add_header X-Content-Type-Options "nosniff"      always;
    add_header X-XSS-Protection      "1; mode=block" always;

    # ── Locations ─────────────────────────────
    {% for location in item.locations | default([{'path': '/', 'type': 'static'}]) %}
    location {{ location.path }} {
        {% if location.type | default('static') == 'proxy' %}
        proxy_pass         {{ location.backend }};
        proxy_http_version 1.1;
        proxy_set_header   Upgrade $http_upgrade;
        proxy_set_header   Connection 'upgrade';
        proxy_set_header   Host $host;
        proxy_set_header   X-Real-IP $remote_addr;
        proxy_cache_bypass $http_upgrade;
        {% elif location.type == 'php' %}
        try_files          $uri =404;
        fastcgi_pass       unix:/run/php/php8.1-fpm.sock;
        fastcgi_index      index.php;
        include            fastcgi_params;
        fastcgi_param      SCRIPT_FILENAME $realpath_root$fastcgi_script_name;
        {% else %}
        try_files          $uri $uri/ =404;
        {% endif %}
    }
    {% endfor %}

    # ── Deny hidden files ─────────────────────
    location ~ /\. {
        deny all;
    }
}
{% if nginx_ssl_enabled and item.ssl | default(false) %}

server {
    listen      443 ssl{{ ' http2' if nginx_http2_enabled else '' }};
    server_name {{ item.domains | join(' ') }};
    root        {{ item.root | default(nginx_web_root) }};

    ssl_certificate     {{ nginx_ssl_cert }};
    ssl_certificate_key {{ nginx_ssl_key }};
    ssl_protocols       TLSv1.2 TLSv1.3;
    ssl_ciphers         HIGH:!aNULL:!MD5;

    # ── HTTP → HTTPS redirect ─────────────────
    if ($scheme != "https") {
        return 301 https://$host$request_uri;
    }

    {% for location in item.locations | default([{'path': '/', 'type': 'static'}]) %}
    location {{ location.path }} {
        try_files $uri $uri/ =404;
    }
    {% endfor %}
}
{% endif %}
EOF
```

## Βήμα 15: Playbook που χρησιμοποιεί το Role

```bash
cat > ~/ansible/playbooks/nginx-role-demo.yml << 'EOF'
---
# ============================================================
# Nginx Role Demo Playbook
# ============================================================

- name: Deploy Nginx με Role
  hosts: "{{ target | default('all_managed') }}"
  become: true
  gather_facts: true

  pre_tasks:
    - name: Deployment info
      ansible.builtin.debug:
        msg:
          - "━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━"
          - "Host : {{ inventory_hostname }}"
          - "OS   : {{ ansible_facts['distribution'] }}"
          - "CPUs : {{ ansible_facts['processor_vcpus'] }}"
          - "RAM  : {{ (ansible_facts['memtotal_mb'] / 1024) | round(1) }}GB"
          - "━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━"
      tags: always

  roles:
    - role: nginx
      vars:
        # Performance (auto αν δεν οριστούν)
        nginx_worker_processes:   "{{ ansible_facts['processor_vcpus'] }}"
        nginx_worker_connections: 1024
        nginx_gzip_enabled:       true

        # Virtual hosts
        nginx_vhosts:
          - name:    main
            domains:
              - "{{ inventory_hostname }}"
              - "{{ ansible_facts['default_ipv4']['address'] }}"
            port:    80
            root:    /var/www/html
            locations:
              - path: /
                type: static
              - path: /api
                type: proxy
                backend: "http://localhost:3000"

  post_tasks:
    - name: Final verification
      ansible.builtin.uri:
        url:         "http://{{ ansible_facts['default_ipv4']['address'] }}"
        status_code: [200, 301, 302]
      register:      final_check
      ignore_errors: true
      tags: verify

    - name: Deployment Report
      ansible.builtin.debug:
        msg:
          - "━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━"
          - "✅ NGINX ROLE DEPLOYMENT COMPLETE"
          - "━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━"
          - "Host   : {{ inventory_hostname }}"
          - "IP     : {{ ansible_facts['default_ipv4']['address'] }}"
          - "Status : {{ '✅ Online' if not final_check.failed else '⚠️ Check manually' }}"
          - "Workers: {{ ansible_facts['processor_vcpus'] }}"
          - "━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━"
      tags: always
EOF
```

    -v

# ── Μόνο install ──────────────────────────────
ansible-playbook playbooks/nginx-role-demo.yml \
    --tags install \
    --limit nextcloud

# ── Μόνο config ───────────────────────────────
ansible-playbook playbooks/nginx-role-demo.yml \
    --tags config \
    --limit nextcloud

# ── Override vars ─────────────────────────────
ansible-playbook playbooks/nginx-role-demo.yml \
    --limit nextcloud \
    -e "nginx_worker_processes=2 nginx_gzip_enabled=false"

# Έλεγχος nginx status

    -a "nginx -t" --become
