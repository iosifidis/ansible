# 26.2 Εγκατάσταση AWX 🔧

## Prerequisites

```
Ελάχιστες απαιτήσεις AWX:
├── 4 CPU cores
├── 8 GB RAM
├── 20 GB disk
├── Docker / Podman
└── Docker Compose v2
```

---

## Εγκατάσταση με Docker Compose (AWX-operator free method)

```yaml
# Playbook: install-awx.yml
---
- name: Install AWX
  hosts: awx_server
  become: true
  gather_facts: true

  vars:
    awx_version:     "26.9.0"
    awx_pg_password: "{{ vault_awx_pg_pass }}"
    awx_secret_key:  "{{ vault_awx_secret_key }}"
    awx_data_dir:    /opt/awx

  tasks:

    - name: Εγκατάσταση prerequisites
      ansible.builtin.package:
        name:
          - docker.io
          - docker-compose-v2
          - git
          - python3-pip
        state: present

    - name: Δομή φακέλων
      ansible.builtin.file:
        path:  "{{ item }}"
        state: directory
        mode:  '0755'
      loop:
        - "{{ awx_data_dir }}"
        - "{{ awx_data_dir }}/postgres"
        - "{{ awx_data_dir }}/redis"
        - "{{ awx_data_dir }}/projects"

    - name: Docker Compose file
      ansible.builtin.template:
        src:  templates/awx-docker-compose.yml.j2
        dest: "{{ awx_data_dir }}/docker-compose.yml"
        mode: '0640'
      no_log: true

    - name: Start AWX stack
      community.docker.docker_compose_v2:
        project_src: "{{ awx_data_dir }}"
        state:       present
        pull:        always

    - name: Wait for AWX
      ansible.builtin.uri:
        url:         "http://localhost:80/api/v2/ping/"
        status_code: 200
      retries: 30
      delay:   10
      until:   not awx_ping.failed
      register: awx_ping

    - name: "✅ AWX Ready!"
      ansible.builtin.debug:
        msg: "AWX διαθέσιμο στο: http://{{ ansible_host }}:80"
```

## AWX Docker Compose template

```yaml
# templates/awx-docker-compose.yml.j2
---
version: '3.8'

x-awx-common: &awx-common
  image: "quay.io/ansible/awx:{{ awx_version }}"
  depends_on:
    - postgres
    - redis
  environment:
    AWX_ADMIN_USER:     admin
    AWX_ADMIN_PASSWORD: "{{ vault_awx_admin_pass }}"
    SECRET_KEY:         "{{ awx_secret_key }}"
    DATABASE_HOST:      postgres
    DATABASE_PORT:      "5432"
    DATABASE_NAME:      awx
    DATABASE_USER:      awx
    DATABASE_PASSWORD:  "{{ awx_pg_password }}"
    REDIS_HOST:         redis
    REDIS_PORT:         "6379"
  volumes:
    - "{{ awx_data_dir }}/projects:/var/lib/awx/projects:rw"

services:

  web:
    <<: *awx-common
    container_name: awx_web
    ports:
      - "80:8052"
    command: /usr/bin/launch_awx.sh

  task:
    <<: *awx-common
    container_name: awx_task
    command: /usr/bin/launch_awx_task.sh

  postgres:
    container_name: awx_postgres
    image:          postgres:15-alpine
    volumes:
      - "{{ awx_data_dir }}/postgres:/var/lib/postgresql/data:rw"
    environment:
      POSTGRES_DB:       awx
      POSTGRES_USER:     awx
      POSTGRES_PASSWORD: "{{ awx_pg_password }}"

  redis:
    container_name: awx_redis
    image:          redis:7-alpine
    volumes:
      - "{{ awx_data_dir }}/redis:/data:rw"
```

## AWX CLI (awxkit)

```bash
# Εγκατάσταση
pip install awxkit

# Ρύθμιση
export TOWER_HOST=http://localhost:80
export TOWER_USERNAME=admin
export TOWER_PASSWORD=mypassword

# Βασικές εντολές
awx organizations list
awx projects list
awx job_templates list
awx jobs list

# Εκτέλεση job
awx job_templates launch \
    --name "Deploy MyApp" \
    --extra_vars '{"app_version": "2.5.0"}'
```
