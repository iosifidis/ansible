# 26.6 Πρακτικό παράδειγμα — AWX API 🚀

## Trigger AWX job από GitHub Actions

```yaml
# .github/workflows/deploy.yml
name: Deploy on Push

on:
  push:
    branches: [main]

jobs:
  deploy:
    runs-on: ubuntu-latest
    steps:

      - name: Trigger AWX Job
        env:
          AWX_HOST:   ${{ secrets.AWX_HOST }}
          AWX_TOKEN:  ${{ secrets.AWX_TOKEN }}
          TEMPLATE_ID: ${{ secrets.DEPLOY_TEMPLATE_ID }}
          APP_VERSION: ${{ github.sha }}
        run: |
          curl -s -X POST \
            "$AWX_HOST/api/v2/job_templates/$TEMPLATE_ID/launch/" \
            -H "Authorization: Bearer $AWX_TOKEN" \
            -H "Content-Type: application/json" \
            -d "{\"extra_vars\": {\"app_version\": \"$APP_VERSION\"}}"

      - name: Wait for job
        env:
          AWX_HOST:  ${{ secrets.AWX_HOST }}
          AWX_TOKEN: ${{ secrets.AWX_TOKEN }}
        run: |
          # Πάρε το job ID από την προηγούμενη εκτέλεση
          JOB_ID=$(curl -s "$AWX_HOST/api/v2/jobs/?order_by=-id&page_size=1" \
            -H "Authorization: Bearer $AWX_TOKEN" | jq '.results[0].id')

          # Περίμενε μέχρι να τελειώσει
          while true; do
            STATUS=$(curl -s "$AWX_HOST/api/v2/jobs/$JOB_ID/" \
              -H "Authorization: Bearer $AWX_TOKEN" | jq -r '.status')
            echo "Job status: $STATUS"
            [ "$STATUS" = "successful" ] && exit 0
            [ "$STATUS" = "failed" ] && exit 1
            sleep 10
          done
```

## Ansible playbook που τρέχει job template

```yaml
---
# trigger-awx-deploy.yml

- name: Trigger AWX Deployment
  hosts: localhost
  gather_facts: false

  vars:
    awx_host:     "{{ lookup('env', 'AWX_HOST') }}"
    awx_token:    "{{ lookup('env', 'AWX_TOKEN') }}"
    template_name: "Deploy MyApp"

  tasks:

    - name: Βρες template ID
      ansible.builtin.uri:
        url:     "{{ awx_host }}/api/v2/job_templates/?name={{ template_name | urlencode }}"
        headers:
          Authorization: "Bearer {{ awx_token }}"
        return_content: true
      register: template_info

    - name: Launch job
      ansible.builtin.uri:
        url:    "{{ awx_host }}/api/v2/job_templates/{{ template_info.json.results[0].id }}/launch/"
        method: POST
        headers:
          Authorization: "Bearer {{ awx_token }}"
        body_format: json
        body:
          extra_vars:
            app_version: "{{ app_version }}"
        status_code: 201
      register: job_launch

    - name: Wait for completion
      ansible.builtin.uri:
        url:     "{{ awx_host }}/api/v2/jobs/{{ job_launch.json.id }}/"
        headers:
          Authorization: "Bearer {{ awx_token }}"
        return_content: true
      register: job_status
      until:    job_status.json.status in ['successful', 'failed', 'error']
      retries:  60
      delay:    10

    - name: Job result
      ansible.builtin.fail:
        msg: "AWX Job failed: {{ job_status.json.status }}"
      when: job_status.json.status != 'successful'
```
