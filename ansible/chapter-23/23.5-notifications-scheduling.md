# 23.5 Notifications & Scheduling 🔔

## Notifications

```yaml
tasks:

  - name: Slack notification template
    ansible.builtin.uri:
      url:    "{{ awx_host }}/api/v2/notification_templates/"
      method: POST
      user:   "{{ awx_user }}"
      password: "{{ awx_pass }}"
      force_basic_auth: true
      body_format: json
      body:
        name:              "Slack Alerts"
        organization:      "{{ org_id }}"
        notification_type: slack
        notification_configuration:
          url:      "{{ vault_slack_webhook }}"
          channels: ["#deployments"]
      status_code: [200, 201]
    no_log: true

  - name: Email notification template
    ansible.builtin.uri:
      url:    "{{ awx_host }}/api/v2/notification_templates/"
      method: POST
      user:   "{{ awx_user }}"
      password: "{{ awx_pass }}"
      force_basic_auth: true
      body_format: json
      body:
        name:              "Email Alerts"
        organization:      "{{ org_id }}"
        notification_type: email
        notification_configuration:
          host:       "{{ smtp_host }}"
          port:       587
          username:   "{{ smtp_user }}"
          password:   "{{ vault_smtp_pass }}"
          use_tls:    true
          recipients: ["admin@example.com"]
          sender:     "awx@example.com"
      status_code: [200, 201]
    no_log: true
```

## Scheduling

```yaml
tasks:

  - name: Weekly patching schedule
    ansible.builtin.uri:
      url:    "{{ awx_host }}/api/v2/schedules/"
      method: POST
      user:   "{{ awx_user }}"
      password: "{{ awx_pass }}"
      force_basic_auth: true
      body_format: json
      body:
        name:    "Weekly Patching"
        enabled: true
        rrule:   "DTSTART:20240101T020000Z RRULE:FREQ=WEEKLY;BYDAY=SU"
        unified_job_template: "{{ patching_template_id }}"
      status_code: [200, 201]
    no_log: true
```
