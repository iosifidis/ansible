# 11.6 Πρακτικά Templates 🛠️

## Τι θα φτιάξουμε

```
templates/
├── sshd_config.j2      ← SSH config (και στις 3 distros)
├── ferm.conf.j2        ← Firewall (Debian)
├── fail2ban_jail.j2    ← Fail2ban (universal)
└── motd.j2             ← Message of the Day (universal)
```

Και ένα **playbook** που τα χρησιμοποιεί όλα!

---

## Template 1: sshd_config.j2

```jinja
{# templates/sshd_config.j2 #}
{# ============================================================
   SSH Server Configuration
   Generated by Ansible for {{ inventory_hostname }}
   OS: {{ ansible_facts['distribution'] }} {{ ansible_facts['distribution_version'] }}
   ============================================================ #}

{# ── Υπολογισμός settings βάσει OS ──────── #}
{% if ansible_facts['os_family'] == 'Debian' %}
{% set sshd_log = '/var/log/auth.log' %}
{% elif ansible_facts['os_family'] == 'RedHat' %}
{% set sshd_log = '/var/log/secure' %}
{% else %}
{% set sshd_log = '/var/log/messages' %}
{% endif %}

# ── Network ───────────────────────────────────────────────────
Port                    {{ ssh_port | default(2022) }}
AddressFamily           any
ListenAddress           0.0.0.0
ListenAddress           ::

# ── Authentication ────────────────────────────────────────────
PermitRootLogin         {{ ssh_permit_root | default('no') }}
PasswordAuthentication  {{ ssh_password_auth | default('no') }}
PubkeyAuthentication    yes
AuthorizedKeysFile      .ssh/authorized_keys

{% if ssh_allowed_users is defined and ssh_allowed_users | length > 0 %}
AllowUsers              {{ ssh_allowed_users | join(' ') }}
{% endif %}

{% if ssh_allowed_groups is defined and ssh_allowed_groups | length > 0 %}
AllowGroups             {{ ssh_allowed_groups | join(' ') }}
{% endif %}

# ── Security ──────────────────────────────────────────────────
MaxAuthTries            {{ ssh_max_auth_tries | default(3) }}
MaxSessions             {{ ssh_max_sessions   | default(10) }}
LoginGraceTime          {{ ssh_login_grace    | default('30s') }}
StrictModes             yes
IgnoreRhosts            yes
HostbasedAuthentication no
PermitEmptyPasswords    no
ChallengeResponseAuth   no
UsePAM                  yes
X11Forwarding           {{ 'yes' if ssh_x11_forwarding | default(false) else 'no' }}
PrintMotd               {{ 'yes' if ssh_print_motd | default(true) else 'no' }}

# ── Timeouts ──────────────────────────────────────────────────
ClientAliveInterval     {{ ssh_alive_interval | default(300) }}
ClientAliveCountMax     {{ ssh_alive_count    | default(3) }}

# ── Logging ───────────────────────────────────────────────────
SyslogFacility          AUTH
LogLevel                {{ ssh_log_level | default('INFO') }}

# ── Subsystems ────────────────────────────────────────────────
{% if ansible_facts['os_family'] == 'RedHat' %}
Subsystem sftp /usr/libexec/openssh/sftp-server
{% else %}
Subsystem sftp /usr/lib/openssh/sftp-server
{% endif %}

{% if ssh_banner is defined %}
# ── Banner ────────────────────────────────────────────────────
Banner                  /etc/ssh/banner
{% endif %}
```

## Template 2: ferm.conf.j2

```jinja
{# templates/ferm.conf.j2 #}
{# ============================================================
   Ferm Firewall Configuration
   Generated by Ansible for {{ inventory_hostname }}
   ============================================================ #}

{# ── Default ports αν δεν ορίζονται ──────── #}
{% set ssh_p    = ssh_port    | default(2022) %}
{% set http_p   = http_port   | default(80)   %}
{% set https_p  = https_port  | default(443)  %}

table filter {

    # ── INPUT chain ───────────────────────────────────────────
    chain INPUT {
        policy DROP;

        # Established connections
        mod state state (ESTABLISHED RELATED) ACCEPT;

        # Loopback
        interface lo ACCEPT;

        # ICMP (ping)
        proto icmp {
            icmp-type (echo-request echo-reply) ACCEPT;
        }

        # SSH
        proto tcp dport {{ ssh_p }} {
{%- if firewall_ssh_rate_limit | default(true) %}
            mod recent name SSH set;
            mod recent name SSH rcheck
                seconds 60 hitcount {{ firewall_ssh_hitcount | default(4) }}
                DROP;
{%- endif %}
            ACCEPT;
        }

{%- if firewall_allow_http | default(true) %}
        # HTTP
        proto tcp dport {{ http_p }} ACCEPT;
{%- endif %}

{%- if firewall_allow_https | default(true) %}
        # HTTPS
        proto tcp dport {{ https_p }} ACCEPT;
{%- endif %}

{%- if custom_ports is defined and custom_ports | length > 0 %}
        # Custom ports
{%- for port in custom_ports %}
        proto {{ port.proto | default('tcp') }} dport {{ port.port }} ACCEPT;
        {#- ↑ custom port rule #}
{%- endfor %}
{%- endif %}

{%- if trusted_ips is defined and trusted_ips | length > 0 %}
        # Trusted IPs — full access
{%- for ip in trusted_ips %}
        saddr {{ ip }} ACCEPT;
{%- endfor %}
{%- endif %}

    } # end INPUT

    # ── OUTPUT chain ──────────────────────────────────────────
    chain OUTPUT {
        policy ACCEPT;
    }

    # ── FORWARD chain ─────────────────────────────────────────
    chain FORWARD {
        policy DROP;
    }

} # end table filter

{%- if firewall_nat | default(false) %}
table nat {
    chain POSTROUTING {
        policy ACCEPT;
        # NAT rules
{%- for rule in nat_rules | default([]) %}
        saddr {{ rule.src }} daddr {{ rule.dst }} MASQUERADE;
{%- endfor %}
    }
}
{%- endif %}
```

## Template 3: fail2ban_jail.j2

```jinja
{# templates/fail2ban_jail.j2 #}
{# ============================================================
   Fail2ban Configuration
   Generated by Ansible for {{ inventory_hostname }}
   OS: {{ ansible_facts['distribution'] }}
   ============================================================ #}

[DEFAULT]
# ── Timing ────────────────────────────────────────────────────
bantime         = {{ fail2ban_bantime  | default(3600) }}
findtime        = {{ fail2ban_findtime | default(600)  }}
maxretry        = {{ fail2ban_maxretry | default(5)    }}

# ── Backend ───────────────────────────────────────────────────
backend         = systemd

# ── Ignore IPs ────────────────────────────────────────────────
{% set ignore_ips = ['127.0.0.1/8', '::1'] %}
{% if fail2ban_ignore_ips is defined %}
{% set ignore_ips = ignore_ips + fail2ban_ignore_ips %}
{% endif %}
ignoreip        = {{ ignore_ips | join(' ') }}

# ── Action ────────────────────────────────────────────────────
banaction        = {{ fail2ban_banaction | default('iptables-multiport') }}
action           = %(action_mwl)s

# ── Email (αν ορίζεται) ───────────────────────────────────────
{% if fail2ban_email is defined %}
destemail       = {{ fail2ban_email }}
sendername      = Fail2ban on {{ inventory_hostname }}
mta             = sendmail
{% endif %}

# ══════════════════════════════════════════════════════════════
# JAILS
# ══════════════════════════════════════════════════════════════

# ── SSH ───────────────────────────────────────────────────────
[sshd]
enabled         = true
port            = {{ ssh_port | default(2022) }}
{% if ansible_facts['os_family'] == 'Debian' %}
logpath         = %(sshd_log)s
{% elif ansible_facts['os_family'] == 'RedHat' %}
logpath         = /var/log/secure
{% else %}
logpath         = /var/log/messages
{% endif %}
maxretry        = {{ fail2ban_ssh_maxretry | default(3) }}
bantime         = {{ fail2ban_ssh_bantime  | default(7200) }}

{% if fail2ban_jails is defined %}
{# ── Custom jails ─────────────────────────── #}
{% for jail in fail2ban_jails %}
# ── {{ jail.name }} ───────────────────────────────────────────
[{{ jail.name }}]
enabled         = {{ 'true' if jail.enabled | default(true) else 'false' }}
port            = {{ jail.port }}
logpath         = {{ jail.logpath }}
maxretry        = {{ jail.maxretry | default(fail2ban_maxretry | default(5)) }}
{% if jail.bantime is defined %}
bantime         = {{ jail.bantime }}
{% endif %}

{% endfor %}
{% endif %}

{% if firewall_allow_http | default(true) %}
# ── Nginx ─────────────────────────────────────────────────────
[nginx-http-auth]
enabled         = true
port            = {{ http_port | default(80) }},{{ https_port | default(443) }}
logpath         = /var/log/nginx/error.log
maxretry        = {{ fail2ban_nginx_maxretry | default(6) }}

[nginx-limit-req]
enabled         = true
port            = {{ http_port | default(80) }},{{ https_port | default(443) }}
logpath         = /var/log/nginx/error.log
maxretry        = {{ fail2ban_nginx_maxretry | default(10) }}
{% endif %}
```

## Το Playbook που τα χρησιμοποιεί

```yaml
---
# templates-deploy.yml
# Deploy όλων των config templates

- name: Deploy Configuration Templates
  hosts: "{{ target | default('all_managed') }}"
  become: true
  gather_facts: true

  vars:
    # ── SSH ───────────────────────────────────
    ssh_port:            2022
    ssh_permit_root:     "no"
    ssh_password_auth:   "no"
    ssh_allowed_users:
      - ansible
    ssh_max_auth_tries:  3
    ssh_alive_interval:  300
    ssh_alive_count:     3
    ssh_print_motd:      true

    # ── Firewall ──────────────────────────────
    firewall_allow_http:   true
    firewall_allow_https:  true
    firewall_ssh_rate_limit: true
    firewall_ssh_hitcount: 4
    custom_ports: []
    trusted_ips:  []

    # ── Fail2ban ──────────────────────────────
    fail2ban_bantime:    3600
    fail2ban_findtime:   600
    fail2ban_maxretry:   5
    fail2ban_ignore_ips: []
    fail2ban_jails:      []

    # ── App ───────────────────────────────────
    app_env:    production
    server_role: managed_server

  tasks:

    # ══════════════════════════════════════════
    # SSH CONFIGURATION
    # ══════════════════════════════════════════

    - name: Deploy sshd_config
      ansible.builtin.template:
        src:      templates/sshd_config.j2
        dest:     /etc/ssh/sshd_config
        owner:    root
        group:    root
        mode:     '0600'
        backup:   true
        validate: sshd -t -f %s
      notify: Restart SSH

    # ══════════════════════════════════════════
    # FIREWALL CONFIGURATION
    # ══════════════════════════════════════════

    - name: Deploy ferm config (Debian)
      ansible.builtin.template:
        src:    templates/ferm.conf.j2
        dest:   /etc/ferm/ferm.conf
        owner:  root
        group:  adm
        mode:   '0640'
        backup: true
      when: ansible_facts['os_family'] == "Debian"
      notify: Restart ferm

    # ══════════════════════════════════════════
    # FAIL2BAN CONFIGURATION
    # ══════════════════════════════════════════

    - name: Deploy fail2ban config
      ansible.builtin.template:
        src:    templates/fail2ban_jail.j2
        dest:   /etc/fail2ban/jail.local
        owner:  root
        group:  root
        mode:   '0644'
        backup: true
      notify: Restart fail2ban

    # ══════════════════════════════════════════
    # MOTD
    # ══════════════════════════════════════════

    - name: Deploy MOTD
      ansible.builtin.template:
        src:   templates/motd.j2
        dest:  /etc/motd
        owner: root
        group: root
        mode:  '0644'

    # ══════════════════════════════════════════
    # FLUSH & VERIFY
    # ══════════════════════════════════════════

    - name: Flush handlers
      ansible.builtin.meta: flush_handlers

    - name: Έλεγχος SSH service
      ansible.builtin.service:
        name:  "{{ service_ssh }}"
        state: started
      register: ssh_check

    - name: Τελικό report
      ansible.builtin.debug:
        msg:
          - "━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━"
          - "✅ Templates deployed!"
          - "━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━"
          - "Host    : {{ inventory_hostname }}"
          - "SSH port: {{ ssh_port }}"
          - "Firewall: ✅"
          - "Fail2ban: ✅"
          - "MOTD    : ✅"
          - "━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━"

  # ══════════════════════════════════════════
  # HANDLERS
  # ══════════════════════════════════════════
  handlers:

    - name: Restart SSH
      ansible.builtin.service:
        name:  "{{ service_ssh }}"
        state: restarted
      ignore_errors: "{{ ansible_check_mode }}"

    - name: Restart ferm
      ansible.builtin.service:
        name:  ferm
        state: restarted
      when: ansible_facts['os_family'] == "Debian"
      ignore_errors: "{{ ansible_check_mode }}"

    - name: Restart fail2ban
      ansible.builtin.service:
        name:  fail2ban
        state: restarted
      ignore_errors: "{{ ansible_check_mode }}"
```

## Εκτέλεση

```bash
# Δημιουργία templates directory
mkdir -p ~/ansible/templates

# Δημιουργία template αρχείων
# (αντιγραφή κώδικα από πάνω)

# Syntax check
ansible-playbook playbooks/templates-deploy.yml \
    --syntax-check

# Dry run — δες τι θα αλλάξει
ansible-playbook playbooks/templates-deploy.yml \
    --check --diff \
    --limit nextcloud

# Εκτέλεση
ansible-playbook playbooks/templates-deploy.yml \
    --limit nextcloud -v

# Μόνο MOTD
ansible-playbook playbooks/templates-deploy.yml \
    --limit nextcloud \
    --tags motd
```
