# 14.5 Πρακτικό παράδειγμα Tags 🛠️

## Τι θα φτιάξουμε

Ένα **πλήρες playbook** με hybrid tag strategy που καλύπτει και τις 3 διανομές:

```
site.yml
├── Tags: install, config, deploy, verify, security
├── Components: nginx, ssh, firewall, fail2ban, app
├── Special: always, never
└── CI/CD ready
```

---

## Το Playbook

```bash
vim ~/ansible/playbooks/site.yml
```

```yaml
---
# ============================================================
# site.yml — Full Stack Playbook με Tags
# Hybrid tag strategy: action + component
# ============================================================

- name: Full Stack Server Setup
  hosts: "{{ target | default('all_managed') }}"
  become: true
  gather_facts: true

  vars:
    app_name:    mywebapp
    app_version: "2.1.4"
    app_env:     production
    http_port:   80
    https_port:  443
    ssh_port:    2022

    common_packages:
      - curl
      - vim
      - git
      - htop
      - tree

  # ══════════════════════════════════════════
  # PRE-TASKS — τρέχουν ΠΑΝΤΑ πρώτα
  # ══════════════════════════════════════════
  pre_tasks:

    - name: Εμφάνιση deployment info
      ansible.builtin.debug:
        msg:
          - "━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━"
          - "Host    : {{ inventory_hostname }}"
          - "OS      : {{ ansible_facts['distribution'] }}"
          - "App     : {{ app_name }} v{{ app_version }}"
          - "Env     : {{ app_env }}"
          - "━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━"
      tags: always    # ← ΠΑΝΤΑ εμφανίζεται

    - name: Validate environment
      ansible.builtin.assert:
        that:
          - app_env in ['production', 'staging', 'development']
          - app_version is defined
        fail_msg: "Invalid environment or missing version!"
      tags: always    # ← ΠΑΝΤΑ ελέγχεται

  # ══════════════════════════════════════════
  # TASKS
  # ══════════════════════════════════════════
  tasks:

    # ──────────────────────────────────────
    # SYSTEM UPDATE
    # Tags: update, packages
    # ──────────────────────────────────────
    - name: System update block
      tags: [update, packages]
      block:

        - name: Update apt (Debian)
          ansible.builtin.apt:
            update_cache: true
            upgrade:      safe
          when: ansible_facts['os_family'] == "Debian"

        - name: Update dnf (RedHat)
          ansible.builtin.dnf:
            name:  "*"
            state: latest
          when: ansible_facts['os_family'] == "RedHat"

        - name: Update zypper (openSUSE)
          community.general.zypper:
            name:  "*"
            state: latest
          when: ansible_facts['os_family'] == "Suse"

    # ──────────────────────────────────────
    # COMMON PACKAGES
    # Tags: install, packages, common
    # ──────────────────────────────────────
    - name: Install common packages block
      tags: [install, packages, common]
      block:

        - name: Install common (Debian)
          ansible.builtin.apt:
            name:  "{{ common_packages }}"
            state: present
          when: ansible_facts['os_family'] == "Debian"

        - name: Install common (RedHat)
          ansible.builtin.dnf:
            name:  "{{ common_packages }}"
            state: present
          when: ansible_facts['os_family'] == "RedHat"

        - name: Install common (openSUSE)
          community.general.zypper:
            name:  "{{ common_packages }}"
            state: present
          when: ansible_facts['os_family'] == "Suse"

    # ──────────────────────────────────────
    # NGINX
    # Tags: install/config/service + nginx
    # ──────────────────────────────────────
    - name: Install nginx (Debian)
      ansible.builtin.apt:
        name:         nginx
        state:        present
        update_cache: true
      when: ansible_facts['os_family'] == "Debian"
      tags: [install, nginx]

    - name: Install nginx (RedHat)
      ansible.builtin.dnf:
        name:  nginx
        state: present
      when: ansible_facts['os_family'] == "RedHat"
      tags: [install, nginx]

    - name: Install nginx (openSUSE)
      community.general.zypper:
        name:  nginx
        state: present
      when: ansible_facts['os_family'] == "Suse"
      tags: [install, nginx]

    - name: Configure nginx
      ansible.builtin.template:
        src:      templates/nginx.conf.j2
        dest:     /etc/nginx/nginx.conf
        validate: nginx -t -c %s
        backup:   true
        mode:     '0644'
      tags: [config, nginx]
      notify: Restart nginx

    - name: Deploy vhost config
      ansible.builtin.template:
        src:  templates/vhost.conf.j2
        dest: "/etc/nginx/sites-available/{{ app_name }}.conf"
        mode: '0644'
      tags: [config, nginx, vhost]
      notify: Reload nginx

    - name: Enable vhost
      ansible.builtin.file:
        src:   "/etc/nginx/sites-available/{{ app_name }}.conf"
        dest:  "/etc/nginx/sites-enabled/{{ app_name }}.conf"
        state: link
      tags: [config, nginx, vhost]
      notify: Reload nginx

    - name: Start nginx
      ansible.builtin.service:
        name:    nginx
        state:   started
        enabled: true
      tags: [service, nginx]

    # ──────────────────────────────────────
    # SECURITY
    # Tags: security + component
    # ──────────────────────────────────────

    # ── SSH ───────────────────────────────
    - name: Configure SSH
      ansible.builtin.template:
        src:      templates/sshd_config.j2
        dest:     /etc/ssh/sshd_config
        validate: sshd -t -f %s
        backup:   true
        mode:     '0600'
      tags: [security, config, ssh]
      notify: Restart SSH

    # ── Firewall ──────────────────────────
    - name: Configure ferm (Debian)
      ansible.builtin.template:
        src:    templates/ferm.conf.j2
        dest:   /etc/ferm/ferm.conf
        backup: true
        mode:   '0640'
      when: ansible_facts['os_family'] == "Debian"
      tags: [security, config, firewall, ferm]
      notify: Restart ferm

    - name: Configure firewalld (RedHat/openSUSE)
      ansible.posix.firewalld:
        port:      "{{ ssh_port }}/tcp"
        permanent: true
        state:     enabled
        immediate: true
      when: ansible_facts['os_family'] in ["RedHat", "Suse"]
      tags: [security, config, firewall]

    # ── Fail2ban ──────────────────────────
    - name: Configure fail2ban
      ansible.builtin.template:
        src:    templates/fail2ban_jail.j2
        dest:   /etc/fail2ban/jail.local
        backup: true
        mode:   '0644'
      tags: [security, config, fail2ban]
      notify: Restart fail2ban

    - name: Start fail2ban
      ansible.builtin.service:
        name:    fail2ban
        state:   started
        enabled: true
      tags: [security, service, fail2ban]

    # ──────────────────────────────────────
    # APPLICATION
    # Tags: deploy/config + app
    # ──────────────────────────────────────

    - name: Create app directories
      ansible.builtin.file:
        path:  "{{ item }}"
        state: directory
        mode:  '0755'
        owner: www-data
        group: www-data
      loop:
        - "/opt/{{ app_name }}"
        - "/opt/{{ app_name }}/logs"
        - "/etc/{{ app_name }}"
      tags: [deploy, app, directories]

    - name: Deploy app config
      ansible.builtin.template:
        src:   templates/app.conf.j2
        dest:  "/etc/{{ app_name }}/app.conf"
        mode:  '0644'
        owner: www-data
      tags: [config, deploy, app]

    - name: Deploy application files
      ansible.builtin.copy:
        src:   "files/{{ app_name }}/"
        dest:  "/opt/{{ app_name }}/"
        owner: www-data
        group: www-data
        mode:  '0644'
      tags: [deploy, app]
      notify: Restart app

    - name: Update app version fact
      ansible.builtin.copy:
        content: "{{ app_version }}\n"
        dest:    "/opt/{{ app_name }}/VERSION"
      tags: [deploy, app]

    # ──────────────────────────────────────
    # SYSTEM CONFIG
    # Tags: config + system
    # ──────────────────────────────────────

    - name: Configure timezone
      community.general.timezone:
        name: "{{ system_timezone | default('Europe/Athens') }}"
      tags: [config, system, timezone]

    - name: Deploy MOTD
      ansible.builtin.template:
        src:  templates/motd.j2
        dest: /etc/motd
        mode: '0644'
      tags: [config, system, motd]

    # ──────────────────────────────────────
    # MONITORING
    # Tags: monitoring (never by default)
    # ──────────────────────────────────────

    - name: Setup monitoring agent
      ansible.builtin.debug:
        msg: "Installing monitoring agent..."
      tags:
        - monitoring
        - never        # ← μόνο αν ζητηθεί!

    # ──────────────────────────────────────
    # VERIFICATION
    # Tags: verify + component
    # ──────────────────────────────────────

    - name: Flush handlers πριν verify
      ansible.builtin.meta: flush_handlers
      tags: [verify, always]

    - name: Verify nginx
      ansible.builtin.uri:
        url:         "http://localhost:{{ http_port }}"
        status_code: 200
      register:     nginx_check
      ignore_errors: true
      tags: [verify, nginx]

    - name: Verify SSH
      ansible.builtin.wait_for:
        port:    "{{ ssh_port }}"
        host:    localhost
        timeout: 10
      tags: [verify, ssh]

    - name: Verify fail2ban
      ansible.builtin.service_facts:
      tags: [verify, fail2ban]

    # ──────────────────────────────────────
    # DEBUG — never by default
    # Tags: debug, never
    # ──────────────────────────────────────

    - name: Debug — Full hostvars
      ansible.builtin.debug:
        var: hostvars[inventory_hostname]
      tags:
        - debug
        - never

    - name: Debug — All facts
      ansible.builtin.debug:
        var: ansible_facts
      tags:
        - debug
        - facts
        - never

    # ──────────────────────────────────────
    # FINAL REPORT
    # Tags: always (τρέχει πάντα)
    # ──────────────────────────────────────

    - name: Final deployment report
      ansible.builtin.debug:
        msg:
          - "━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━"
          - "✅ DEPLOYMENT COMPLETE"
          - "━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━"
          - "Host   : {{ inventory_hostname }}"
          - "App    : {{ app_name }} v{{ app_version }}"
          - "Nginx  : {{ '✅' if nginx_check is defined
                         and not nginx_check.failed
                         else '⚠️ Check manually' }}"
          - "━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━"
      tags: always    # ← ΠΑΝΤΑ εμφανίζεται

  # ══════════════════════════════════════════
  # POST-TASKS
  # ══════════════════════════════════════════
  post_tasks:

    - name: Save deployment log
      ansible.builtin.copy:
        dest: "/var/log/ansible_deploy_{{ ansible_facts['date_time']['date'] }}.log"
        content: |
          Deployment Log
          Date   : {{ ansible_facts['date_time']['iso8601'] }}
          Host   : {{ inventory_hostname }}
          App    : {{ app_name }} v{{ app_version }}
          Env    : {{ app_env }}
          Ansible: {{ ansible_version.full }}
        mode: '0644'
      tags: always

  # ══════════════════════════════════════════
  # HANDLERS
  # ══════════════════════════════════════════
  handlers:

    - name: Restart nginx
      ansible.builtin.service:
        name:  nginx
        state: restarted
      ignore_errors: "{{ ansible_check_mode }}"

    - name: Reload nginx
      ansible.builtin.service:
        name:  nginx
        state: reloaded
      ignore_errors: "{{ ansible_check_mode }}"

    - name: Restart SSH
      ansible.builtin.service:
        name:  "{{ service_ssh }}"
        state: restarted
      ignore_errors: "{{ ansible_check_mode }}"

    - name: Restart ferm
      ansible.builtin.service:
        name:  ferm
        state: restarted
      when: ansible_facts['os_family'] == "Debian"
      ignore_errors: "{{ ansible_check_mode }}"

    - name: Restart fail2ban
      ansible.builtin.service:
        name:  fail2ban
        state: restarted
      ignore_errors: "{{ ansible_check_mode }}"

    - name: Restart app
      ansible.builtin.service:
        name:  "{{ app_name }}"
        state: restarted
      ignore_errors: true
```

## Εκτέλεση — Όλα τα Σενάρια

```bash
# ── Προεπισκόπηση tags ────────────────────────
ansible-playbook playbooks/site.yml --list-tags
ansible-playbook playbooks/site.yml --list-tasks --tags nginx

# ── Full deployment ───────────────────────────
ansible-playbook playbooks/site.yml \
    --limit nextcloud

# ── Dry run ───────────────────────────────────
ansible-playbook playbooks/site.yml \
    --check --diff \
    --limit nextcloud

# ── Μόνο εγκαταστάσεις ───────────────────────
ansible-playbook playbooks/site.yml \
    --tags install \
    --limit nextcloud

# ── Μόνο config ───────────────────────────────
ansible-playbook playbooks/site.yml \
    --tags config \
    --limit nextcloud

# ── Μόνο security ─────────────────────────────
ansible-playbook playbooks/site.yml \
    --tags security \
    --limit nextcloud

# ── Μόνο nginx ────────────────────────────────
ansible-playbook playbooks/site.yml \
    --tags nginx \
    --limit nextcloud

# ── Deploy μόνο app ───────────────────────────
ansible-playbook playbooks/site.yml \
    --tags deploy \
    --limit nextcloud \
    -e "app_version=2.2.0"

# ── Verify μόνο ───────────────────────────────
ansible-playbook playbooks/site.yml \
    --tags verify \
    --limit nextcloud

# ── Skip updates (γρήγορο) ────────────────────
ansible-playbook playbooks/site.yml \
    --skip-tags update \
    --limit nextcloud

# ── Debug mode ────────────────────────────────
ansible-playbook playbooks/site.yml \
    --tags "always,debug" \
    --limit nextcloud

# ── CI/CD Pipeline ────────────────────────────
for stage in install config deploy verify; do
    echo "━━━ Stage: ${stage} ━━━"
    ansible-playbook playbooks/site.yml \
        --tags "${stage}" \
        --limit nextcloud
done

# ── Monitoring (never by default) ─────────────
ansible-playbook playbooks/site.yml \
    --tags monitoring \
    --limit nextcloud
```
